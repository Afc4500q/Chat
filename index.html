async function loadDashboard() { try { // استرجاع معرف المستخدم من LocalStorage const userId = localStorage.getItem('userId'); if (!userId) { showError('لم يتم العثور على معرف المستخدم. يرجى تسجيل الدخول مرة أخرى.'); return; } // جلب البيانات بناءً على معرف المستخدم const response = await axios.get(`https://673a394a339a4ce445179ae7.mockapi.io/User/${userId}`); const userData = response.data; // عرض المعلومات الشخصية document.getElementById('user-info').innerHTML = `

الاسم الكامل

${userData.name}

اسم المستخدم

${userData.user}

الرصيد المتاح

${userData.total} IQD

`; // عرض الإحصائيات document.getElementById('stats').innerHTML = `
الطلبات الكلية

${userData.total}

الطلبات الناجحة

${userData.successful}

الطلبات المعلقة

${userData.pending}

الطلبات المرفوضة

${userData.rejected}

`; // عرض المنتجات const productsList = document.getElementById('products-list'); productsList.innerHTML = ''; // مسح المحتوى القديم userData.products.forEach(product => { productsList.innerHTML += `
${product.name}
رقم الهاتف
${product.phone || 'غير متوفر'}
السعر
${product.price} IQD
الحالة
${product.status}
الولاية
${product.state}
المدينة
${product.city}
المنطقة
${product.area}
النوع: ${product.type}
`; }); } catch (error) { console.error('حدث خطأ في جلب البيانات:', error); showError('تعذر تحميل البيانات، يرجى المحاولة لاحقاً'); } }
